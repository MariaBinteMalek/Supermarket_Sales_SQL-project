--============================================================================================================================================================
SELECT
SUM(Total) AS total_sales,
SUM(cogs) AS total_cost,
SUM(Quantity) AS total_quantity,
SUM(`gross income`) As total_gross_income
FROM `super-market-sales-data`;

--==========================================================================================================================================================
--2.City-wise Sales Analysis
--==========================================================================================================================================================
SELECT
City,
SUM(Total) AS total_sales,
SUM(Quantity) AS total_quantity
FROM `super-market-sales-data`
GROUP BY City
ORDER BY SUM(Total) DESC;

--==========================================================================================================================================================
--18.Advance top Product line by city
--==========================================================================================================================================================
WITH city_product_sales AS (
SELECT
City,
`Product line`,
SUM(Total) AS Total_Sales
FROM `super-market-sales-data`
GROUP BY City, 'Product line`
),
ranked_product AS (
SELECT
City,
`Product line`,
Total_Sales,
RANK() OVER(
PARTITION BY City
ORDER BY Total_Sales DESC
) AS Sales_Rank
FROM city_product_sales
)
SELECT
City,
`Product line`,
`Total_Sales`,
Sales_Rank
FROM ranked_products
WHERE Sales_Rank=1
ORDER BY City;
--==========================================================================================================================================================
--19.Advance running total
--==========================================================================================================================================================
SELECT
Date,
SUM(Total) AS Daily_Sales,
SUM(SUM(Total)) OVER (
ORDER BY Date
ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
) AS Running_Total_Sales
FROM `super-market-sales-data`
GROUP BY Date
ORDER BY Date;
--==========================================================================================================================================================
--20.LAG Previous Sales
--==========================================================================================================================================================
SELECT
`Invoice ID`,
`Date`,
`Total`,
LAG(`Total`) OVER (ORDER BY `DATE','Invoice ID`)AS Previous_Sales
FROM `super-market-sales-data`
ORDER BY `Date', 'Invoice ID`;

--==========================================================================================================================================================
--21.LEAD NEXT Sales
--==========================================================================================================================================================
SELECT
`Invoice ID`,
`Date`,
`Total`,
LEAD(`Total`) OVER (ORDER BY `DATE','Invoice ID`)AS next_sales
FROM `super-market-sales-data`
ORDER BY `Date', 'Invoice ID`;
--==========================================================================================================================================================
--22. LAG_LEAD_Combined.png
--==========================================================================================================================================================
SELECT
`Invoice ID`,
`Date`,
`Total`,
LAG(`Total`) OVER (ORDER BY `DATE','Invoice ID`)AS Previous_sales,
LEAD(`Total`) OVER (ORDER BY `DATE','Invoice ID`)AS next_sales
FROM `super-market-sales-data`
ORDER BY `Date', 'Invoice ID`;
